        /** Run the state-advance loop; returns true if any step signed something (drove a vote/promise). */
        private async runAdvanceLoop(record: TrxRecord<TTerms, TPayload>): Promise<boolean> {
            let acted = false
            while (await this.advanceState(record)) {
                acted = true
            }
            return acted
        }

        /**
         * Arm or clear our deadline timer after adopting a record. Members only (a pure relay holds no
         * stake). A settled/released record clears any timer; a still-`pending` one arms (or keeps) a
         * one-shot timer at the transaction code's own expiry (decision 1) + our skew grace (decision 6).
         */
        private async updateDeadlineTimer(record: TrxRecord<TTerms, TPayload>): Promise<void> {
            if (!this.memberByKey(record.topology, this.self.key)) {
                return
            }
            // NOTE: one extra classification per update (recordOutcome re-verifies commits). Fine at lift
            // scale; if voter sets grow, cache verification (see countVerifiedCommits NOTE).
            const settled = (await this.recordOutcome(record)) !== 'pending'
            if (settled || (await this.store.isReleased(record.transactionCode))) {
                this.clearDeadline(record.transactionCode)
                return
            }
            this.scheduleDeadlineTick(record)
        }

        /**
         * Arm (or leave armed) the one-shot deadline timer for a non-terminal record. Idempotent: a code
         * that already has a live timer keeps it (never stack one per delivery). The deadline is the
         * transaction code's embedded expiry (decision 1) judged against OUR clock (decision 6); it fires a
         * `skewGraceMs` grace past expiry so a modestly-fast clock does not prematurely void.
         *
         * We do NOT arm once the code is already expired at `now`: the expired-code path (NC/NP) already
         * runs inline in `advanceState` during the very update that reaches here, so a zero-delay timer
         * would only re-run a no-op — and, worse, re-arm in a loop as onDeadline's push re-delivers (see
         * design.md). One-shot in 6.1; **SEAM for `tallynet-stalled-recovery` (6.2)**: the periodic
         * re-push/pull retry loop re-arms here on each fire.
         */
        private scheduleDeadlineTick(record: TrxRecord<TTerms, TPayload>): void {
            const transactionCode = record.transactionCode
            if (this.deadlineTimers.has(transactionCode)) {
                return
            }
            let expiry: number
            try {
                expiry = this.sessionCodes.expiresAt(transactionCode)
            } catch {
                return // a malformed code cannot be scheduled against (validation should already have refused it)
            }
            const now = this.clock.now()
            if (now >= expiry) {
                return // deadline already passed; the expired path runs inline, so a timer adds nothing (6.2 owns retry)
            }
            const handle = this.clock.setTimer(() => {
                this.deadlineTimers.delete(transactionCode) // one-shot: spent the moment it fires
                return this.onDeadline(transactionCode)
            }, Math.max(0, expiry + this.skewGraceMs - now))
            this.deadlineTimers.set(transactionCode, handle)
        }

        /** Cancel and forget the deadline timer for `transactionCode` (idempotent). */
        private clearDeadline(transactionCode: string): void {
            const handle = this.deadlineTimers.get(transactionCode)
            if (handle === undefined) {
                return
            }
            this.deadlineTimers.delete(transactionCode)
            this.clock.clearTimer(handle)
        }

        /**
         * The deadline fired (or a test/host drove it): reload the stored record and, if we are still a
         * member of a non-terminal, unreleased lift, re-run the advance loop now the code has expired.
         * The expired-code paths already in `advanceState` do the work — an un-voted voter over a COMPLETE
         * promise set votes `NC` (void-on-timeout, decision 3); a holding-but-un-promised participant
         * signs `NP` (wake-from-sleep). A node with nothing to do signs nothing and stays `pending`
         * (decision 5: no unilateral void from silence). One-shot: does NOT re-arm (6.2 owns retry).
         * Independently callable so a unit test can drive the timeout against a seeded store.
         */
        async onDeadline(transactionCode: string): Promise<void> {
            const record = await this.store.getRecord(transactionCode)
            if (!record || !this.memberByKey(record.topology, this.self.key) || (await this.store.isReleased(transactionCode))) {
                this.clearDeadline(transactionCode)
                return
            }
            const acted = await this.runAdvanceLoop(record)
            this.traceSink({ ...this.traceBase(record), type: 'deadline-reached', transactionCode, acted })
            await this.push(record)
            if ((await this.recordOutcome(record)) !== 'pending') {
                this.clearDeadline(transactionCode) // a timeout NC may have settled us
            }
        }

        interface ArmedTimer { fn: () => void | Promise<void>; delayMs: number; cleared: boolean }
        interface MutableClock extends Clock {
            set(now: number): void
            /** Live (armed, not cleared) timers, in arm order. */
            armed: ArmedTimer[]
            /** Count of live timers. */
            liveTimers(): number
            /** Fire every live timer once, in arm order, advancing nothing (a test sets `now` first). */
            fireAll(): Promise<void>
        }
        function makeClock(start = NOW): MutableClock {
            let current = start
            const armed: ArmedTimer[] = []
            const clock: MutableClock = {
                now: () => current,
                setTimer: (fn, ms) => {
                    const t: ArmedTimer = { fn, delayMs: ms, cleared: false }
                    armed.push(t)
                    return t
                },
                clearTimer: (handle) => {
                    const t = handle as ArmedTimer | undefined
                    if (t && armed.includes(t)) { t.cleared = true }
                },
                set: (next) => { current = next },
                armed,
                liveTimers: () => armed.filter((t) => !t.cleared).length,
                fireAll: async () => {
                    for (const t of armed) {
                        if (!t.cleared) { await t.fn() }
                    }
                },
            }
            return clock
        }

§G — WHY (verified by tracing; do NOT re-derive, just trust + implement)
════════════════════════════════════════════════════════════════════════════════════════════════════
• **onDeadline's voter-NC branch is unreachable through normal delivery.** `advanceState`'s
  `ourCommitNeeded` case votes synchronously inside the same `update` that completes the promise phase, so
  a voter is NEVER left holding a COMPLETE promise set un-voted. Every existing stall is therefore an
  onDeadline no-op (even-split: both already voted; dropped/silent: holders incomplete or `[P]`). ⇒ cover
  the NC branch by SEEDING the store (E2a), not via update.

• **The re-arm infinite-loop hazard — and the fix.** `VirtualClock.runUntilIdle`/`advance` advance
  `virtualNow` to each timer's scheduled time as it fires, so a deadline timer at `expiresAt+grace` makes
  the code EXPIRED when it fires. onDeadline → push → neighbour `update` → if that update re-armed a timer
  while `now>=expiry`, the delay would be 0 → it fires again → push → … → `runUntilIdle` throws at
  MAX_STEPS. FIX (C8 `scheduleDeadlineTick`): **do not arm when `now >= expiry`.** Combined with `push`'s
  existing skip-test (a peer whose last-acked copy equals what we'd push is skipped — and after normal
  propagation every peer already holds the final record), onDeadline's push delivers nothing new, no
  node re-arms, and the heap drains. Verified against every pending-leaving e2e test.

• **Determinism holds** (`consensus.test.ts` ~1053). Commit/void runs settle before the deadline instant,
  so every node CLEARS its timer before it could fire ⇒ no `deadline-reached` events and no extra fires in
  those runs; arm/clear emit no trace. Two seeded runs assign identical `seq`s ⇒ identical fire order ⇒
  byte-identical streams. (The straddle test F2 is new/isolated and advances virtual time to ~1.2M ms —
  fine for the virtual clock.)

• **Validation still rejects an expired code at FIRST sight** (`isValid` fails) — so a brand-new expired
  record is refused, never adopted, never arms a timer. A record we ALREADY hold that expires is the
  timer's job. The two paths stay distinct; test both (E2a = held; validation test already covers first-sight).

• **No PPV timer (decision 4), no unilateral void (decision 5), no ring/star node (decision 7), trust only
  our clock (decision 6), timers in-memory only (restart re-arm = documented seam).** Nothing to build for
  these beyond the design.md prose in §H.

The first of the propagation-timeout work (split from the plan ticket `tallynet-propagation-timeouts`).
Sibling tickets: `tallynet-stalled-recovery` (seq 6.2, `prereq`s this) adds the periodic re-push/pull loop;
`tallysim-timeout-scenarios` (seq 6.3) drives and measures both in the simulator.

Today TallyNet consensus has **no timer**. The state machine (`participant.ts`) advances only when a record
is delivered (a push or a pull reply). A fully-promised lift whose voters never finish voting — a voter that
stays silent, an even-split tie, a dropped commit push — sits in the `propagating` state forever, with every
participant's promised credit held. This ticket gives the lift a deadline and the one in-protocol action a
stalled lift can safely take on its own: **drive un-voted voters to vote void (`NC`) at the deadline**, so a
voter majority (commit *or* void) is always eventually reached and resources release.

## The decisions this ticket builds (resolved in planning — write them into design.md)

**1. The deadline is the transaction code's embedded expiry — a single lift-wide instant. No new field.**
It becomes an explicit proposal parameter two ways: the originator *chooses* it via optional `ttlMs?` on
`UniQuery` (and `proceed` mints from `args.ttlMs ?? args.query.ttlMs ?? DEFAULT_TTL`, clamped to the
session-code ceiling); a node *judges* it via a protocol pre-gate in `promiseDecision` (refuse → `NP` when
`expiresAt(transactionCode) - now < minPromiseMarginMs`). The margin is per-node (principle 0), with
`MIN_PROMISE_MARGIN_MS` as default only.

**2. One deadline, not per-phase.** Which action at the deadline is structural (what we have signed), not
temporal: an un-voted voter in the commit phase → `NC`; an un-promised participant holding the record →
`NP` (rare in practice). A slow promise phase eats the voters' window — the originator sizes `ttlMs`; a node
with too little left refuses to promise.

**3. Void-on-timeout is a *voter* `NC`, commit-phase only.** A `C`/`NC` preimage binds the complete promise
set (`digest.ts`), so an `NC` over an incomplete set would not aggregate. Available only once
`promisePhaseComplete`. Pre-completion void is an `NP` (binds only the core; any `NP` ⇒ `rejected`).

**4. The pre-promise-void *timer* is a no-op in the immediate-promise model — do not build one.** A
participant promises synchronously on first sight; it never lingers un-promised across time. Keep the
`anyCodeExpired → NP` path (fires when an expired record is first *held*, e.g. wake-from-sleep).

**5. No unilateral void from silence.** A node concludes `void` only on evidence (an `NP`, or an `NC`
majority). Timing out while waiting never by itself settles — it stays `pending`.

**6. Clock skew: trust only your own clock (v1).** Arm from `expiresAt` vs our own `clock.now()`, fire at
`expiresAt + SKEW_GRACE_MS`. Gross skew is a liveness risk only; the void stays atomic. The ChipNet
"reject a peer NC dated before the deadline" rule needs signed timestamps — note it, build nothing.

**7. Ring/star convergence is NOT adopted.** Push-to-all-reachable + contact points + the pull already
give every cooperating member the outcome; the 6.2 recovery loop subsumes ChipNet's ring routing.

## Edge cases & interactions (all still apply — the blueprint's tests cover them)
- Voter already voted `C`, deadline passes → `propagating`, no NC (E2b). MUST NOT equivocate.
- Promise phase incomplete, we hold `P` → `promising`; tick signs nothing (E2c).
- Even-split tie reached before deadline → both voted → deadline no-op → stays `pending` (documented limit).
- `NC` majority vs a late `C` after settlement → release latches; late `C` cannot flip (idempotent release).
- Threshold above 1/2 → un-voted voters vote `NC` pushing toward void; a method that can reach neither side
  stays `pending` (originator owns choosing a reachable method).
- Expired code at FIRST sight → rejected at validation, no timer armed. A HELD record that expires → the
  timer's job. Both paths tested.
- `SKEW_GRACE_MS` < `MIN_PROMISE_MARGIN_MS` so a node that promised still has margin to vote.
- Timer cleared on terminal + abort + release; no leaked timers.
- Determinism: all timing through the injected `Clock`; `no-node-globals.test` stays green.
